### Reproduction source

Save the following as `/tmp/mu3lab-review-reproduce.py` and run from the reviewed repository root with its dependencies installed:

```bash
PYTHONPATH=. /tmp/mu3lab-review-venv/bin/python /tmp/mu3lab-review-reproduce.py
```

Only fake test values appear below. External calls are mocked; SQLite writes use temporary directories. The script sets its temporary runtime root before importing project modules. It creates one `mu3lab-review-isolated-*` temporary directory for imported runtime defaults; it is disposable review data.

```python
import importlib.util
import json
import os
import tempfile
from pathlib import Path
from unittest.mock import MagicMock, patch

os.environ['MU3LAB_RUNTIME_ROOT'] = tempfile.mkdtemp(prefix='mu3lab-review-isolated-')
from ctl.jobs import JobStore
from ctl.lifecycle.app_releases import Release, write, from_history
from ctl.runtime import RuntimePaths
from ctl.api.security import resolve_identity, csrf_token, mutation_allowed
from starlette.requests import Request

root = Path.cwd()
spec = importlib.util.spec_from_file_location('review_gateway', root / 'platform/tool-gateway/gateway.py')
g = importlib.util.module_from_spec(spec)
spec.loader.exec_module(g)
app = {'name':'Demo','server_id':'demo','categories':[{'id':'data','enabled':True}], 'tools': {'edit': {'name':'edit','category':'data','access':'write','core':True,'enabled':True}}, 'blocked':{}, 'upstream':{'url':'http://invalid'}}
connector = MagicMock()
connector.list_tools.return_value = {'edit': {'name':'edit'}}
connector.request.return_value = {'content':[]}
with patch.object(g.CONNECTORS, 'get', return_value=connector):
    g.handle_call('demo', app, 'change_with_tool', {'tool':'edit','arguments':{'value':1}})
    g.handle_call('demo', app, 'edit', {'value':2})
print('E01 gateway writes executed without approval:', connector.request.call_count)

client = g.Connector({'url':'http://invalid'})
client.session = 'existing-session'
with patch.object(client, '_post', side_effect=[OSError('reply lost after side effect'), ({'result': {'content':[]}}, '')]) as post:
    client.request('tools/call', {'name':'edit','arguments':{}})
    print('E02 same write submitted after ambiguous transport failure:', post.call_count)

with tempfile.TemporaryDirectory() as folder:
    p = Path(folder) / 'policy.json'
    p.write_text(json.dumps({'apps':{'demo':app}}))
    policy = g.PolicyStore(str(p))
    policy.get()
    p.unlink()
    print('E03 deleted policy still authorizes cached app:', 'demo' in policy.get()['apps'])

with tempfile.TemporaryDirectory() as folder:
    store = JobStore(Path(folder)/'db.sqlite')
    a = store.create(kind='lifecycle',service_id='demo',action='start',actor='alice',idempotency_key='same')
    b = store.create(kind='lifecycle',service_id='other',action='stop',actor='bob',idempotency_key='same')
    c = store.create(kind='lifecycle',service_id='demo',action='stop',actor='alice',idempotency_key='different')
    print('E04 key reused by different actor/action/service returns first job:',a['id']==b['id'])
    print('E05 conflicting action on active service returns first job:',a['id']==c['id'])

with tempfile.TemporaryDirectory() as folder:
    paths=RuntimePaths(Path(folder))
    (paths.projects/'demo').mkdir(parents=True)
    write('demo',Release('1.0',{'db':'db@sha256:old'}),paths)
    write('demo',Release('1.0',{'db':'db@sha256:new'}),paths)
    print('E06 previous same-version release overwritten:',from_history('demo','1.0',paths).images['db'])

headers={'host':'demo.ts.net:8446','origin':'https://demo.ts.net:8446','x-authentik-username':'claimed-admin','x-authentik-uid':'claimed-subject','x-authentik-groups':'mu3lab-operators','x-authentik-jwt':'caller-chosen','x-mu3lab-proxy-token':'proxy-injected'}
def req(h):
    return Request({'type':'http','method':'GET','scheme':'https','path':'/','headers':[(k.encode(),v.encode()) for k,v in h.items()]})
with patch('ctl.api.security._runtime_env',return_value={'MU3LAB_INGRESS_TOKEN':'proxy-injected','MU3LAB_CTL_TOKEN':'control-test'}),patch('ctl.api.security._record_operator_traversal'):
    identity=resolve_identity(req(headers))
    headers['x-mu3lab-csrf']=csrf_token(req(headers))
    print('E07 bootstrap-proxy-equivalent headers accepted as admin and mutation:', identity['is_admin'],mutation_allowed(req(headers)))
print('All reproductions used temporary stores and mocked upstream calls; no live services accessed.')

from ctl.registry import load
from ctl.service_state import status
service = load().get('litellm')
with patch('ctl.service_state._healthy',return_value=(True,'healthy')), patch('ctl.service_state.route_answers',return_value=False):
    result = status(service,'test.ts.net',root,{service.private_https_port},{str(service.compose_path(root).resolve()):'running'})
    print('E08 failed route probe:', {k:result[k] for k in ['state','route_state','route_ready']})
```
